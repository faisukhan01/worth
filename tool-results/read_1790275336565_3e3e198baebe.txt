    19→interface TimelineEntry {
    20→  ts: string
    21→  event: string
    22→  detail: string

    24→
    25→interface IncidentFilters {
    26→  status: string
    27→  severity: string
    28→  service: string
    29→  assignee: string
    30→}
    31→
    32→const NO_FILTERS: IncidentFilters = { status: '', severity: '', service: '', assignee: '' }
    33→
    34→function filtersActive(f: IncidentFilters): boolean {
    35→  return Boolean(f.status || f.severity || f.service || f.assignee)
    36→}
    37→
    38→function filterLabel(f: IncidentFilters): string {
    39→  const parts: string[] = []
    40→  if (f.status) parts.push(`status:${f.status}`)
    41→  if (f.severity) parts.push(`sev:${f.severity}`)
    42→  if (f.service) parts.push(f.service)
    43→  if (f.assignee === '__none') parts.push('unassigned')
    44→  else if (f.assignee) parts.push(`@${f.assignee.split(' ')[0].toLowerCase()}`)
    45→  return parts.join(' · ') || 'empty'
    46→}

    49→  triggered: 'crit',
    50→  acknowledged: 'warn',
    51→  mitigated: 'neutral',
    52→  resolved: 'ok',
    53→  assignment: 'neutral',
    54→}

   139→      { ruleId: rule.id },

   154→                    label: 'Fire drill',

   246→  }

   250→      <div className="flex flex-wrap items-center justify-between gap-3">

   254→            {data.counts.open} open · {data.counts.rulesEnabled}/{data.counts.rules} rules armed
   255→            {filtersActive(filters) && (
   256→              <>
   257→                {' '}· <span className="text-foreground/80">{open.length} match filter</span>

   259→            )}
   260→          </p>
   261→        </div>

   268→        <StatCard icon={<BellRing className="h-3.5 w-3.5" />} label="Acknowledged" value={data.counts.acknowledged} filteredValue={filtersActive(filters) ? data.incidents.filter((i) => i.status === 'acknowledged' && matches(i)).length : undefined} tone="warn" />

   270→        <StatCard icon={<ShieldCheck className="h-3.5 w-3.5" />} label="Resolved (register)" value={data.counts.resolved} filteredValue={filtersActive(filters) ? data.incidents.filter((i) => i.status === 'resolved' && matches(i)).length : undefined} tone="ok" />
   271→      </div>
   272→
   273→      <Tabs value={tab} onValueChange={setTab}>

   275→          <TabsTrigger value="incidents" className="text-xs">Incidents</TabsTrigger>
   276→          <TabsTrigger value="rules" className="text-xs">Rules</TabsTrigger>
   277→          <TabsTrigger value="activity" className="gap-1 text-xs">
   278→            <History className="h-3 w-3" /> Activity
   279→          </TabsTrigger>
   280→        </TabsList>
   281→
   282→        <TabsContent value="incidents" className="mt-3 space-y-2">

   284→          {(presets.length > 0 || filtersActive(filters)) && (
   285→            <div className="flex flex-wrap items-center gap-1.5 px-1">
   286→              {presets.map((p) => {
   287→                const active = filterLabel(filters) === filterLabel({ status: p.status, severity: p.severity, service: p.service, assignee: p.assignee })
   288→                return (
   289→                  <span
   290→                    key={p.id}
   291→                    role="button"
   292→                    tabIndex={0}
   293→                    onClick={() => setFilters({ status: p.status, severity: p.severity, service: p.service, assignee: p.assignee })}
   294→                    onKeyDown={(e) => {
   295→                      if (e.key === 'Enter' || e.key === ' ') {
   296→                        e.preventDefault()
   297→                        setFilters({ status: p.status, severity: p.severity, service: p.service, assignee: p.assignee })
   298→                      }
   299→                    }}
   300→                    className={cn(
   301→                      'group inline-flex cursor-pointer items-center gap-1 rounded-full border px-2.5 py-0.5 text-[10px] transition-colors',
   302→                      active
   303→                        ? 'border-primary/50 bg-primary/15 text-foreground'
   304→                        : 'bg-muted/40 text-muted-foreground hover:border-ring/50 hover:text-foreground',
   305→                    )}
   306→                  >
   307→                    <Filter className="h-2.5 w-2.5" />
   308→                    {p.label}

   315→                    </button>
   316→                  </span>
   317→                )
   318→              })}
   319→              {filtersActive(filters) && (
   320→                <button
   321→                  className="text-[10px] text-muted-foreground underline-offset-2 hover:text-foreground hover:underline"
   322→                  onClick={() => setFilters(NO_FILTERS)}
   323→                >
   324→                  clear
   325→                </button>
   326→              )}
   327→            </div>
   328→          )}
   329→

   331→          <div className="card-surface flex flex-wrap items-center gap-2 px-3 py-2">
   332→            <Filter className="h-3 w-3 shrink-0 text-muted-foreground" />
   333→            <Select value={filters.status} onValueChange={(v) => setFilters({ ...filters, status: v === '__all' ? '' : v })}>
   334→              <SelectTrigger className="h-7 w-[7.5rem] text-[11px]"><SelectValue placeholder="Status" /></SelectTrigger>

   336→                <SelectItem value="__all" className="text-xs">any status</SelectItem>
   337→                {['triggered', 'acknowledged', 'mitigated', 'resolved'].map((s) => (
   338→                  <SelectItem key={s} value={s} className="text-xs capitalize">{s}</SelectItem>
   339→                ))}
   340→              </SelectContent>
   341→            </Select>
   342→            <Select value={filters.severity} onValueChange={(v) => setFilters({ ...filters, severity: v === '__all' ? '' : v })}>
   343→              <SelectTrigger className="h-7 w-[7.5rem] text-[11px]"><SelectValue placeholder="Severity" /></SelectTrigger>

   387→            >
   388→              <Pin className="h-3 w-3" /> Pin filter
   389→            </Button>
   390→          </div>
   391→
   392→          {/* Selection + bulk action bar */}
   393→          <div className="flex flex-wrap items-center gap-2 px-1">

   401→              {checked.size > 0 ? `${checked.size} selected` : `${openIds.length} open · select for bulk actions`}

   408→                <Button size="sm" variant="outline" className="h-7 text-[11px]" disabled={bulk.isPending} onClick={() => bulkAct('mitigate')}>
   409→                  Mitigate
   410→                </Button>
   411→                <Button size="sm" variant="outline" className="h-7 text-[11px] text-ok hover:text-ok" disabled={bulk.isPending} onClick={() => bulkAct('resolve')}>
   412→                  Resolve

   415→                  <SelectTrigger className="h-7 w-40 gap-1 text-[11px]" aria-label="Bulk assign to team member">
   416→                    <UserPlus className="h-3 w-3 text-muted-foreground" />
   417→                    <SelectValue placeholder="Assign to…" />

   434→          </div>
   435→          {[...open, ...resolvedShown].map((i) => {
